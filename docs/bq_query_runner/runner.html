bq_query_runner.runner

bq_query_runner - run SQL scripts on Google BigQuery from files.

This is the single entry-point module: it holds the helper functions, the RunQuery and Logger classes and the main() function wired to the bq-query-runner console script.

Arguments are read EITHER from the command line OR from a JSON config file (mutually exclusive). Working folders (sql, log, placeholder, output) and relative paths are resolved against the current working directory (see base_dir()), so the tool behaves like a standard CLI once installed.

 96def is_absolute_path(path):
 97    """Return True only for genuinely absolute paths.
 98
 99    On Windows a path is considered absolute only when it carries a drive letter
100    or a UNC anchor: a bare leading slash/backslash (e.g. "/sql") is NOT treated
101    as absolute and must be resolved against the base directory rather than the
102    drive root.
103    """
104    if os.name == 'nt':
105        return bool(os.path.splitdrive(path)[0])
106    return os.path.isabs(path)

109def resolve_paths(input_param):
110    """Resolve relative path parameters against the base directory.
111
112    Drive-less paths that start with a separator (e.g. "/sql" or "\\sql") are
113    treated as relative too: their leading separators are stripped so they
114    resolve to <base_dir>/sql instead of the drive root (C:/sql).
115
116    Args:
117        input_param: dictionary containing all the input arguments.
118
119    Returns:
120        The same dictionary with path parameters resolved to absolute paths.
121    """
122    root = base_dir()
123    for param in PATH_PARAMS:
124        value = input_param.get(param)
125        if not value:
126            continue
127        if is_absolute_path(value):
128            input_param[param] = os.path.normpath(value)
129        else:
130            input_param[param] = os.path.normpath(os.path.join(root, value.lstrip('/\\')))
131    return input_param

134def check_arguments(input_param):
135    """
136    Function to check the input arguments.
137
138    Args:
139        input_param: dictionary containing all the input arguments from main function
140    """
141    try:
142        # Validate project
143        if not input_param.get('PROJECT'):
144            raise Exception("The PROJECT argument is required. Provide it in the command line or in the config file.")
145
146        # Validate input paths: sql_path must exist if provided; placeholder_path
147        # is validated whenever it is set (providing it is what enables substitution).
148        # service_account_json_path is intentionally NOT validated here: a missing service account falls back to default auth.
149        # Output paths (log/output) may not exist yet, as long as the parent does.
150        input_paths = ['sql_path']
151        if input_param.get('placeholder_path'):
152            input_paths.append('placeholder_path')
153        for path_param in input_paths:
154            value = input_param.get(path_param)
155            if value and not os.path.exists(value):
156                raise Exception(f"{path_param} does not exist: {value}")
157
158        for path_param in ['log_path', 'output_path']:
159            value = input_param.get(path_param)
160            if value and not os.path.exists(value):
161                parent = os.path.dirname(value)
162                if parent and not os.path.exists(parent):
163                    raise Exception(f"{path_param} parent directory does not exist: {parent}")
164
165        # Validate replace string format
166        if input_param.get('replace'):
167            replacements = input_param.get('replace').split(',')
168            for replacement in replacements:
169                if ':' not in replacement:
170                    raise Exception(
171                        f"Invalid replacement format: {replacement}. Expected format: original:replacement")
172
173        # Validate output format (covers the config-file source too)
174        output_format = input_param.get('output_format')
175        if output_format and output_format.lower() not in ['parquet', 'csv', 'json']:
176            raise Exception(f"Invalid output_format: {output_format}. Must be one of: parquet, csv, json")
177
178        # Create default work folders only for the parameters that were not provided
179        if not input_param.get('sql_path'):
180            check_work_folders('sql')
181        if not input_param.get('log_path'):
182            check_work_folders('log')
183        if not input_param.get('placeholder_path'):
184            check_work_folders('placeholder')
185        if not input_param.get('output_path'):
186            check_work_folders('output')
187
188    except Exception as e:
189        print(f"E: Error while validating arguments: {str(e)}")
190        sys.exit(1)

333    def load_placeholders(self):
334        """
335        Function to load placeholder definitions from the placeholder file
336        (loaded once and cached). Placeholder substitution is enabled simply by
337        providing ``placeholder_path``: when it is not set, no substitution is done.
338
339        Returns:
340            Dictionary mapping placeholder names to their substitution values
341        """
342        # No file provided -> placeholders are disabled.
343        if not self.placeholder_path:
344            return {}
345
346        # Return the cached placeholders if already loaded
347        if self._placeholders is not None:
348            return self._placeholders
349
350        # Cache the outcome from here on (even an empty dict) so the lookup runs once.
351        self._placeholders = {}
352
353        # Load placeholders from the file
354        if not os.path.isfile(self.placeholder_path):
355            self.logger.log(f"W: Placeholder file not found at {self.placeholder_path}. "
356                            "Continuing without placeholders.")
357            return self._placeholders
358
359        try:
360            with open(self.placeholder_path, 'r', encoding=TEXT_ENCODING) as f:
361                placeholders = json.load(f)
362            self.logger.log(f"I: Loaded {len(placeholders)} placeholders from {self.placeholder_path}")
363            for name, value in placeholders.items():
364                self.logger.log(f"I: Placeholder loaded: {name} = {value}")
365            self._placeholders = placeholders
366        except Exception as e:
367            self.logger.log(f"W: Error loading placeholder file {self.placeholder_path}: {str(e)}. "
368                            "Continuing without placeholders.")
369
370        return self._placeholders

388    def prep_sql_file(self, sql_file):
389        """
390        Function to prepare the SQL file to work, applying placeholder substitution
391        and text replacements if any.
392
393        Args:
394            sql_file: SQL file to prepare
395
396        Returns:
397            List of SQL statements with placeholders and replacements applied
398        """
399        try:
400            # Read the file with a BOM-tolerant UTF-8 codec (prevents
401            # 'Illegal input character' errors on files carrying a UTF-8 BOM).
402            with open(sql_file, "r", encoding=TEXT_ENCODING) as f:
403                data = f.read()
404
405            # Apply ${...} placeholder substitution when a placeholder file is provided
406            if self.placeholder_path:
407                placeholders = self.load_placeholders()
408                if placeholders:
409                    referenced = re.findall(r'\${([^}]+)}', data)
410
411                    # Check that all referenced placeholders are defined
412                    missing = [name for name in referenced if name not in placeholders]
413                    if missing:
414                        self.logger.log(f"E: Missing placeholders: {', '.join(missing)}")
415                        sys.exit(1)
416
417                    for name, value in placeholders.items():
418                        token = "${" + name + "}"
419                        if token in data:
420                            data = data.replace(token, str(value))
421                            self.logger.log(f"I: Placeholder replaced in SQL File - {token}:{value}")
422
423            # Apply replace in the SQL file, if present
424            if self.replace is not None:
425                string_couples = self.replace.split(',')
426                for rep in string_couples:
427                    words_rep = rep.split(':')
428                    if len(words_rep) == 2:
429                        data = data.replace(words_rep[0], words_rep[1])
430                        self.logger.log(f"I: Replace in SQL File - {words_rep[0]}:{words_rep[1]}")
431                    else:
432                        self.logger.log(f"W: Skipping invalid replacement format: {rep}")
433
434            return self.split_sql_statements(data)
435
436        except Exception as e:
437            self.logger.log(f"E: Error working SQL file {sql_file}: {str(e)}")
438            sys.exit(1)

440    def get_size(self, size_bytes):
441        """
442        Function to get readable size of query data processed.
443
444        Args:
445            size_bytes: total bytes processed
446
447        Returns:
448            Readable size of data processed
449        """
450        if not size_bytes:
451            return "0B"
452        size_name = ("B", "KB", "MB", "GB", "TB", "PB", "EB", "ZB", "YB")
453        i = int(math.floor(math.log(size_bytes, 1024)))
454        p = math.pow(1024, i)
455        s = round(size_bytes / p, 2)
456        return f"{s}{size_name[i]}"

458    def start_process(self, client, sql_files, res_content):
459        """
460        Function starting the process.
461
462        For each SQL file, prepare the string and run the queries.
463
464        Args:
465            client: BigQuery client instance
466            sql_files: List of all the SQL files to work
467            res_content: Content from resume file (previous run log)
468        """
469        for sql_file in sql_files:
470            self.logger.log(f"I: Working file: {sql_file}")
471            instructions = self.prep_sql_file(sql_file)
472
473            # run queries
474            for i, instruction in enumerate(instructions):
475                # Check if it is a resume and the query has already been processed
476                query_id = f"{sql_file}:{i+1}"
477                if query_id in res_content:
478                    self.logger.log(f"I: Skipping query {i+1} (already processed)")
479                    continue
480
481                # Log the query we're about to run (first 100 chars)
482                preview = instruction.replace("\n", " ")[:100] + "..."
483                self.logger.log(f"I: Running query {i+1} - {preview}")
484
485                # Configure and execute the query job
486                job_config = bigquery.QueryJobConfig(dry_run=self.dry_run)
487
488                try:
489                    job = client.query(instruction, job_config=job_config)
490
491                    # Wait for the job to complete and check results
492                    result = job.result()
493
494                    # Get data processed and execution time
495                    size = self.get_size(job.total_bytes_processed)
496                    time_info = ""
497                    if not self.dry_run and job.ended and job.started:
498                        time_info = f". Execution time: {job.ended - job.started}"
499                    self.logger.log(f"I: Query completed. Processed: {size}{time_info}")
500
501                    # Save results if requested and there are records to save
502                    if self.output_format and not self.dry_run:
503                        total_rows = getattr(result, 'total_rows', None) or 0
504                        if total_rows > 0:
505                            df = result.to_dataframe()
506                            self.save_results(df, sql_file, i + 1)
507                        else:
508                            self.logger.log(f"W: No records to save for query {i+1}")
509
510                    # Record successful execution for resume functionality
511                    self.logger.log_resume(query_id)
512
513                except Exception:
514                    self.logger.log(f"E: {traceback.format_exc()}")
515                    sys.exit(1)

517    def save_results(self, df, sql_file, stmt_index):
518        """
519        Function to save query results to a file.
520
521        Args:
522            df: DataFrame containing query results
523            sql_file: The SQL file path
524            stmt_index: 1-based index of the statement within the SQL file
525        """
526        try:
527            # Default to the ./output folder if no output_path is provided
528            if self.output_path is None:
529                self.output_path = os.path.join(base_dir(), "output")
530            if not os.path.exists(self.output_path):
531                os.makedirs(self.output_path)
532
533            # Build a filesystem-safe output filename (no path/colon in the name)
534            sql_file_name = os.path.splitext(os.path.basename(sql_file))[0]
535            output_format = self.output_format.lower()
536            file_name = f"{self.logger.timestamp}_{sql_file_name}_stmt{stmt_index}.{output_format}"
537            file_path = os.path.join(self.output_path, file_name)
538
539            # Save based on format
540            if output_format == 'csv':
541                df.to_csv(file_path, index=False)
542            elif output_format == 'json':
543                df.to_json(file_path, orient='records', lines=True)
544            elif output_format == 'parquet':
545                table = pa.Table.from_pandas(df)
546                pq.write_table(table, file_path)
547            else:
548                self.logger.log(f"E: Unsupported output format '{output_format}'. Results not saved.")
549                return
550
551            self.logger.log(f"I: Results saved to {file_path}")
552
553        except Exception as e:
554            self.logger.log(f"E: Error saving results: {str(e)}")
555            sys.exit(1)

477                query_id = f"{sql_file}:{i+1}"
478                if query_id in res_content:
479                    self.logger.log(f"I: Skipping query {i+1} (already processed)")
480                    continue

482                # Log the query we're about to run (first 100 chars)
483                preview = instruction.replace("\n", " ")[:100] + "..."
484                self.logger.log(f"I: Running query {i+1} - {preview}")
485
486                # Configure and execute the query job
487                job_config = bigquery.QueryJobConfig(dry_run=self.dry_run)
488
489                try:
490                    job = client.query(instruction, job_config=job_config)
491
492                    # Wait for the job to complete and check results
493                    result = job.result()
494

496                    size = self.get_size(job.total_bytes_processed)
497                    time_info = ""
498                    if not self.dry_run and job.ended and job.started:
499                        time_info = f". Execution time: {job.ended - job.started}"
500                    self.logger.log(f"I: Query completed. Processed: {size}{time_info}")
501
502                    # Save results if requested and there are records to save
503                    if self.output_format and not self.dry_run:
